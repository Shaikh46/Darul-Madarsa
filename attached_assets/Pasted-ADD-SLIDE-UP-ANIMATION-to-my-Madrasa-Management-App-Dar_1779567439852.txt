// Function to animate dashboard on load
function animateDashboard() {
  const dashboard = document.getElementById('dashboardContent');
  if (dashboard) {
    dashboard.classList.remove('slide-up');
    // Force reflow to restart animation
    void dashboard.offsetWidth;
    dashboard.classList.add('slide-up');
  }
  
  // Animate individual cards with stagger
  const cards = document.querySelectorAll('.dashboard-card, .stat-card, .widget');
  cards.forEach((card, index) => {
    card.style.animation = `slideUp 0.5s ease-out forwards`;
    card.style.animationDelay = `${index * 0.05}s`;
  });
}

// Call after login
function onLoginSuccess() {
  showDashboard();
  animateDashboard();
}

// Call when switching to dashboard from menu
function showDashboard() {
  // ... existing code to show dashboard
  setTimeout(animateDashboard, 50);
}

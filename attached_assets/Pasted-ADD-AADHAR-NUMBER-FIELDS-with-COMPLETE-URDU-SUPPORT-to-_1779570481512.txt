function translateToUrdu() {
  // Translate Aadhar labels
  const studentAadharLabel = document.getElementById('studentAadharLabel');
  if (studentAadharLabel) {
    studentAadharLabel.innerText = "طالب علم کا آدھار نمبر";
  }
  
  const parentAadharLabel = document.getElementById('parentAadharLabel');
  if (parentAadharLabel) {
    parentAadharLabel.innerText = "والدین کا آدھار نمبر";
  }
  
  // Translate hints
  const studentHint = document.getElementById('studentAadharHint');
  if (studentHint) {
    studentHint.innerText = "١٢ ہندسوں کا نمبر - اختیاری";
  }
  
  const parentHint = document.getElementById('parentAadharHint');
  if (parentHint) {
    parentHint.innerText = "١٢ ہندسوں کا نمبر - اختیاری";
  }
  
  // Translate placeholders
  const studentInput = document.getElementById('studentAadhar');
  if (studentInput && studentInput.getAttribute('placeholder') === "1234-5678-9012") {
    studentInput.setAttribute('placeholder', "١٢٣٤-٥٦٧٨-٩٠١٢");
  }
  
  const parentInput = document.getElementById('parentAadhar');
  if (parentInput && parentInput.getAttribute('placeholder') === "1234-5678-9012") {
    parentInput.setAttribute('placeholder', "١٢٣٤-٥٦٧٨-٩٠١٢");
  }
  
  // Translate table headers if they exist
  const tableHeaders = document.querySelectorAll('th');
  tableHeaders.forEach(header => {
    if (header.innerText === "Student Aadhar") {
      header.innerText = "طالب علم آدھار";
    }
    if (header.innerText === "Parent Aadhar") {
      header.innerText = "والدین آدھار";
    }
  });
}

5. AADHAR VALIDATION (Bilingual Messages)
javascript
function validateAadhar(aadhar, type) {
  // Remove spaces and hyphens
  let clean = aadhar.replace(/[\s-]/g, '');
  
  // Empty is allowed
  if (clean.length === 0) return true;
  
  // Check exactly 12 digits
  if (clean.length !== 12) {
    const currentLang = localStorage.getItem('language');
    if (currentLang === 'urdu') {
      alert(`براہ کرم درست ١٢ ہندسوں کا ${type === 'student' ? 'طالب علم' : 'والدین'} آدھار نمبر درج کریں`);
    } else {
      alert(`Please enter valid 12-digit ${type === 'student' ? 'Student' : 'Parent'} Aadhar number`);
    }
    return false;
  }
  
  // Check only digits
  if (!/^\d+$/.test(clean)) {
    const currentLang = localStorage.getItem('language');
    if (currentLang === 'urdu') {
      alert(`غلط آدھار نمبر - صرف ہندسے درج کریں`);
    } else {
      alert(`Invalid Aadhar number - only digits allowed`);
    }
    return false;
  }
  
  return true;
}

// Auto-format Aadhar as user types (works in both languages)
function formatAadhar(input) {
  let value = input.value.replace(/[\s-]/g, '');
  if (value.length >= 4) {
    value = value.slice(0,4) + '-' + value.slice(4);
  }
  if (value.length >= 9) {
    value = value.slice(0,9) + '-' + value.slice(9);
  }
  input.value = value.slice(0, 14);
}

function displayStudents() {
  const students = JSON.parse(localStorage.getItem('students')) || [];
  const tbody = document.getElementById('studentsTableBody');
  tbody.innerHTML = '';
  
  students.forEach((student, index) => {
    const row = tbody.insertRow();
    row.innerHTML = `
      <td class="p-3">${index + 1}</td>
      <td class="p-3">${student.full_name || ''}</td>
      <td class="p-3">${student.class || ''}</td>
      <td class="p-3">${student.father_name || ''}</td>
      <td class="p-3">${student.student_aadhar || '-'}</td>
      <td class="p-3">${student.parent_aadhar || '-'}</td>
      <td class="p-3">${student.phone || ''}</td>
      <td class="p-3">
        <button onclick="editStudent('${student.id}')" class="text-blue-500 mr-2">✏️</button>
        <button onclick="deleteStudent('${student.id}')" class="text-red-500">🗑️</button>
      </td>
    `;
  });
}

// When saving student
function saveStudent() {
  const student = {
    id: document.getElementById('studentId').value || Date.now().toString(),
    full_name: document.getElementById('fullName').value,
    father_name: document.getElementById('fatherName').value,
    mother_name: document.getElementById('motherName').value,
    date_of_birth: document.getElementById('dob').value,
    admission_date: document.getElementById('admissionDate').value,
    class: document.getElementById('class').value,
    phone: document.getElementById('phone').value,
    address: document.getElementById('address').value,
    // NEW AADHAR FIELDS
    student_aadhar: document.getElementById('studentAadhar').value || '',
    parent_aadhar: document.getElementById('parentAadhar').value || '',
    status: 'active'
  };
  
  // Validate Aadhar before saving
  if (!validateAadhar(student.student_aadhar, 'student')) return;
  if (!validateAadhar(student.parent_aadhar, 'parent')) return;
  
  // Save to localStorage
  let students = JSON.parse(localStorage.getItem('students')) || [];
  const existingIndex = students.findIndex(s => s.id === student.id);
  
  if (existingIndex >= 0) {
    students[existingIndex] = student;
  } else {
    students.push(student);
  }
  
  localStorage.setItem('students', JSON.stringify(students));
  displayStudents();
  closeModal();
  
  // Show success message
  const currentLang = localStorage.getItem('language');
  alert(currentLang === 'urdu' ? 'طالب علم محفوظ ہوگیا' : 'Student saved successfully');
}

function editStudent(id) {
  const students = JSON.parse(localStorage.getItem('students')) || [];
  const student = students.find(s => s.id === id);
  
  if (student) {
    document.getElementById('studentId').value = student.id;
    document.getElementById('fullName').value = student.full_name || '';
    document.getElementById('fatherName').value = student.father_name || '';
    document.getElementById('motherName').value = student.mother_name || '';
    document.getElementById('dob').value = student.date_of_birth || '';
    document.getElementById('admissionDate').value = student.admission_date || '';
    document.getElementById('class').value = student.class || '';
    document.getElementById('phone').value = student.phone || '';
    document.getElementById('address').value = student.address || '';
    // LOAD AADHAR VALUES
    document.getElementById('studentAadhar').value = student.student_aadhar || '';
    document.getElementById('parentAadhar').value = student.parent_aadhar || '';
    
    openModal('Edit Student');
  }
}

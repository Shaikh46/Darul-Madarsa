## REQUIREMENT 3: NEXT PRAYER CALCULATION

Auto-calculate and display the next upcoming prayer based on current time:

```javascript
function getNextPrayer() {
  const now = new Date();
  const currentTime = now.getHours() * 60 + now.getMinutes();
  
  const prayers = [
    { name: "Fajr", azanHour: 4, azanMin: 50, jamaatHour: 5, jamaatMin: 15 },
    { name: "Zuhr", azanHour: 13, azanMin: 40, jamaatHour: 14, jamaatMin: 0 },
    { name: "Asr", azanHour: 17, azanMin: 15, jamaatHour: 17, jamaatMin: 30 },
    { name: "Maghrib", azanHour: 18, azanMin: 57, jamaatHour: 18, jamaatMin: 57 },
    { name: "Isha", azanHour: 20, azanMin: 30, jamaatHour: 20, jamaatMin: 45 }
  ];
  
  for (let prayer of prayers) {
    const prayerTime = prayer.azanHour * 60 + prayer.azanMin;
    if (prayerTime > currentTime) {
      return prayer;
    }
  }
  return prayers[0]; // Return Fajr for next day
}

let currentLanguage = 'english';

function toggleLanguage() {
  if (currentLanguage === 'english') {
    translateToUrdu();
    currentLanguage = 'urdu';
    document.getElementById('langText').innerHTML = 'اردو | English';
  } else {
    translateToEnglish();
    currentLanguage = 'english';
    document.getElementById('langText').innerHTML = 'English | اردو';
  }
  localStorage.setItem('language', currentLanguage);
}

function translateToUrdu() {
  // Translate ALL elements with data-urdu attribute or by text content
  document.querySelectorAll('[data-english]').forEach(el => {
    if (el.getAttribute('data-urdu')) {
      el.innerText = el.getAttribute('data-urdu');
    }
  });
  
  // Translate buttons
  document.querySelectorAll('button, .btn').forEach(btn => {
    const englishText = btn.innerText.trim();
    const urduMap = {
      "Add Student": "طالب علم شامل کریں",
      "Add Teacher": "استاد شامل کریں",
      "Record Donation": "عطیہ ریکارڈ کریں",
      "Record Fee": "فیس ریکارڈ کریں",
      "Add Expense": "خرچ شامل کریں",
      "Save": "محفوظ کریں",
      "Cancel": "منسوخ کریں",
      "Edit": "ترمیم کریں",
      "Delete": "حذف کریں",
      "Search": "تلاش کریں"
    };
    if (urduMap[englishText]) {
      btn.innerText = urduMap[englishText];
    }
  });
  
  // Set Urdu font and direction
  document.body.style.fontFamily = "'Noto Nastaliq Urdu', 'Jameel Noori Nastaleeq', serif";
  document.body.style.direction = "rtl";
  document.body.style.textAlign = "right";
  
  // Update sidebar menu items
  updateSidebarToUrdu();
  
  // Update dashboard cards
  updateDashboardToUrdu();
  
  // Store preference
  localStorage.setItem('language', 'urdu');
}

function translateToEnglish() {
  location.reload(); // Simple reload to reset to English
}

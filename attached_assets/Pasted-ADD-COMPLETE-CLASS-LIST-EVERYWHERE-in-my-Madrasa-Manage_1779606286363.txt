URDU TRANSLATION FUNCTION FOR CLASSES:
javascript
function translateClassesToUrdu() {
  // Translate all class dropdowns
  const classSelects = document.querySelectorAll('select[id$="Class"]');
  
  classSelects.forEach(select => {
    Array.from(select.options).forEach(option => {
      const urduValue = option.getAttribute('data-urdu');
      if (urduValue) {
        option.textContent = urduValue;
      }
    });
  });
  
  // Translate optgroup labels
  const optgroups = document.querySelectorAll('optgroup');
  const optgroupTranslations = {
    "📚 Deeniyat Classes": "📚 دینیات کلاسیں",
    "📖 Farsi Classes": "📖 فارسی کلاسیں",
    "🕌 Arabic Classes": "🕌 عربی کلاسیں",
    "🌟 Hifz Classes": "🌟 حفظ کلاسیں"
  };
  
  optgroups.forEach(optgroup => {
    const originalLabel = optgroup.getAttribute('label');
    if (optgroupTranslations[originalLabel]) {
      optgroup.setAttribute('label', optgroupTranslations[originalLabel]);
    }
  });
}

function translateClassesToEnglish() {
  // Revert class dropdowns to English
  const classSelects = document.querySelectorAll('select[id$="Class"]');
  
  classSelects.forEach(select => {
    Array.from(select.options).forEach(option => {
      const englishValue = option.getAttribute('value');
      if (englishValue) {
        option.textContent = englishValue;
      }
    });
  });
  
  // Revert optgroup labels
  const optgroups = document.querySelectorAll('optgroup');
  const optgroupEnglish = {
    "📚 دینیات کلاسیں": "📚 Deeniyat Classes",
    "📖 فارسی کلاسیں": "📖 Farsi Classes",
    "🕌 عربی کلاسیں": "🕌 Arabic Classes",
    "🌟 حفظ کلاسیں": "🌟 Hifz Classes"
  };
  
  optgroups.forEach(optgroup => {
    const urduLabel = optgroup.getAttribute('label');
    if (optgroupEnglish[urduLabel]) {
      optgroup.setAttribute('label', optgroupEnglish[urduLabel]);
    }
  });
}

// Function to translate entire page
function translateToUrdu() {
  // Translate all elements with text
  document.querySelectorAll('*').forEach(element => {
    if (element.childNodes.length === 1 && element.childNodes[0].nodeType === Node.TEXT_NODE) {
      let text = element.innerText;
      for (let [english, urdu] of Object.entries(urduTranslations)) {
        if (text.includes(english)) {
          element.innerText = text.replace(new RegExp(english, 'g'), urdu);
        }
      }
    }
  });
  
  // Translate input placeholders
  document.querySelectorAll('input, textarea').forEach(input => {
    let placeholder = input.getAttribute('placeholder');
    if (placeholder && urduTranslations[placeholder]) {
      input.setAttribute('placeholder', urduTranslations[placeholder]);
    }
  });
  
  // Translate select options
  document.querySelectorAll('select option').forEach(option => {
    if (urduTranslations[option.textContent]) {
      option.textContent = urduTranslations[option.textContent];
    }
  });
  
  // Translate buttons
  document.querySelectorAll('button, a, .btn').forEach(btn => {
    if (urduTranslations[btn.textContent.trim()]) {
      btn.textContent = urduTranslations[btn.textContent.trim()];
    }
  });
  
  // Set Urdu font and direction
  document.body.style.fontFamily = "'Noto Nastaliq Urdu', 'Jameel Noori Nastaleeq', serif";
  document.body.style.direction = "rtl";
  document.body.style.textAlign = "right";
  
  // Store language preference
  localStorage.setItem('language', 'urdu');
}

// Function to translate back to English
function translateToEnglish() {
  location.reload(); // Simple reload to reset to English
}
